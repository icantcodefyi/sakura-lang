Decode command-line flags into a record and count a selected file's bytes.

`Cli.parse` infers `Args` from the binding. `Path` validates the relative path;
`Fs.rooted(Dir.subdir(...))` narrows access to `data` before reading. The manifest
permits reads only below `./data`. Counting bytes avoids treating file contents as
trusted instructions or decoding them unnecessarily. This small recipe reads the
whole file; use `Fs.with_file` and bounded `Fs.read_chunk` for large inputs.

`CliError` and `FsError` propagate to the CLI's failure reporter. Tests replace
arguments and files with `Cli.of` and `Fs.memory`, checking success, missing flags
and a missing file. The harness also runs `sakura run -- --input note.txt` against
the committed data file and expects `4 bytes`.

[Source project](https://github.com/icantcodefyi/sakuralang/tree/main/compiler/examples/cli_files) · harness mode: `run + test`.

### Source and tests

`src/main.sak` includes all required imports and tests.

```sakura test id=cookbook-cli_files project=cli_files
import std/cli.{Cli, CliError}
import std/fs.{Dir, Fs, FsError}
import std/path.{Path}

type Args = { input: Path }

fn byte_count() -> Int ! {Cli, Fs, Fail<CliError | FsError>} {
  let args: Args = Cli.parse(Cli.args())
  with Fs.rooted(Dir.subdir(Fs.root(), "data")) { Bytes.len(Fs.read(args.input)) }
}

pub fn main() -> Unit ! {Cli, Console, Fs, Fail<CliError | FsError>} {
  Console.print("{byte_count()} bytes")
}

test "CLI arguments select a file inside data" {
  with Cli.of(["--input", "note.txt"]),
      Fs.memory([(Path.of("data/note.txt"), String.to_bytes("tea\n"))]) {
    expect_eq(byte_count(), 4)
  }
}

test "missing arguments fail before reading" {
  with Cli.of([]), Fs.memory([]) {
    expect_fails_with(fn() { byte_count() }, fn(e) {
      match e {
        CliError { .. } => true,
        _ => false,
      }
    })
  }
}

test "a missing file remains FsError" {
  with Cli.of(["--input", "missing.txt"]),
      Fs.memory([(Path.of("data/note.txt"), String.to_bytes("tea\n"))]) {
    expect_fails_with(fn() { byte_count() }, fn(e) {
      match e {
        FsError { .. } => true,
        _ => false,
      }
    })
  }
}
```
